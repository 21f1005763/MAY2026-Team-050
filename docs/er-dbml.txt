-- public.contacts definition

-- Drop table

-- DROP TABLE public.contacts;

CREATE TABLE public.contacts (
	id uuid NOT NULL,
	wa_id varchar(32) NOT NULL,
	profile_name varchar(255) NULL,
	created_at timestamptz NOT NULL,
	updated_at timestamptz NOT NULL,
	CONSTRAINT pk_contacts PRIMARY KEY (id),
	CONSTRAINT uq_contacts_wa_id UNIQUE (wa_id)
);

);


-- public.geocode_cache definition

-- Drop table

-- DROP TABLE public.geocode_cache;

CREATE TABLE public.geocode_cache (
	provider varchar(32) NOT NULL,
	lat_round float8 NOT NULL,
	lon_round float8 NOT NULL,
	accept_language varchar(32) NOT NULL,
	status varchar(16) NOT NULL,
	display_address text NULL,
	address_components jsonb NULL,
	raw jsonb NULL,
	created_at timestamptz NOT NULL,
	CONSTRAINT pk_geocode_cache PRIMARY KEY (provider, lat_round, lon_round, accept_language)
);


-- public.jurisdictions definition

CREATE TABLE public.jurisdictions (
	id varchar(64) NOT NULL,
	"name" varchar(255) NOT NULL,
	taxonomy_version varchar(32) NOT NULL,
	is_demo bool DEFAULT true NOT NULL,
	active bool DEFAULT true NOT NULL,
	geofence jsonb NULL,

	updated_at timestamptz NOT NULL,
	CONSTRAINT pk_jurisdictions PRIMARY KEY (id)
);


-- public.webhook_events definition

-- Drop table

-- DROP TABLE public.webhook_events;

CREATE TABLE public.webhook_events (
	id uuid NOT NULL,
	"source" varchar(64) NOT NULL,
	signature_valid bool NOT NULL,
	payload jsonb NOT NULL,
	created_at timestamptz NOT NULL,
	processed_at timestamptz NULL,
	attempt_count int4 NOT NULL,
	locked_until timestamptz NULL,
	CONSTRAINT pk_webhook_events PRIMARY KEY (id)
);
CREATE INDEX ix_webhook_events_created_at ON public.webhook_events USING btree (created_at);
CREATE INDEX ix_webhook_events_source ON public.webhook_events USING btree (source);
